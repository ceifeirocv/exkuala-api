function encodeCursor(startAt: Date, id: string): string {
  return Buffer.from(`${startAt.toISOString()}__${id}`).toString('base64url');
}
function decodeCursor(cursor: string): { cursorStartAt: string; cursorId: string } {
  const raw = Buffer.from(cursor, 'base64url').toString('utf8');
  const [cursorStartAt, cursorId] = raw.split('__');
  return { cursorStartAt, cursorId };
}

    findOwnedById(organizerId: string, eventId: string): Promise<EventResponseDto>

    - returns toResponseDto(event)

    update(organizerId: string, eventId: string, dto: UpdateEventDto): Promise<EventResponseDto>
    - findOwnedOrThrow first
    - If event.status === CANCELLED → throw ConflictException (D-05: frozen)
    - If dto.status present → assertTransitionAllowed(current, target)
    - If transitioning to PUBLISHED → assertPublishGate(event after field updates)
    - Apply field updates (description, venueName, address, title, startAt, endAt, ticketPrice, externalTicketUrl, categoryId)
    - Apply status if present
    - save() and return toResponseDto()

    softDeleteDraft(organizerId: string, eventId: string): Promise<void>
    - findOwnedOrThrow first
    - If event.status !== DRAFT → throw ConflictException (D-15)
    - repository.softDelete(eventId)

    toResponseDto(event: EventEntity): EventResponseDto
    - Manual mapping — no ClassSerializerInterceptor
    - Excludes deletedAt

    private assertTransitionAllowed(current: EventStatus, target: EventStatus): void
    - allowed map: DRAFT→[PUBLISHED], PUBLISHED→[CANCELLED], CANCELLED→[]
    - throws ConflictException if not allowed

    private assertPublishGate(event: EventEntity): void
    - publishRequiredFields: ['title', 'description', 'startAt', 'venueName', 'address', 'categoryId']
    - missing = fields where event[f] is null/undefined
    - if missing.length > 0: throw UnprocessableEntityException({ statusCode: 422, message: 'Cannot publish: missing required fields', missing })
    - if event.startAt <= new Date(): throw UnprocessableEntityException({ statusCode: 422, message: 'Cannot publish: event date has passed', missing: [] })

    private async findOwnedOrThrow(eventId: string, organizerId: string): Promise<EventEntity>
    - findOne({ where: { id: eventId, organizerId } })
